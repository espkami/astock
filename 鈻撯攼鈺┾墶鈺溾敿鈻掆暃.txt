# docker-compose.yml
version: "3.9"

services:
  astock:
    image: ghcr.io/espkami/astock:latest
    container_name: astock_app
    ports:
      - "8000:8000"
    volumes:
      - astock_data:/app/data
    env_file:
      - .env
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/api/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 20s

volumes:
  astock_data:
    driver: local

==================================================================================================================================================================
同目录下建一个 .env 文件：
JWT_SECRET=随便写一串你自己的随机字符
ADMIN_USERNAME=admin
ADMIN_PASSWORD=你想要的密码
